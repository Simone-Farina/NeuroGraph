```typescript
async findByUrl(client: TypedClient, userId: string, url: string): Promise<KnowledgeQueueItem | null> {
  const { data, error } = await client

    .select('*')
    .eq('user_id', userId)
```

**2. Create `src/lib/capture/metadata.ts`:**

Create the directory `src/lib/capture/` and file `metadata.ts` with an `extractHeadMetadata` function:

```typescript
/**
 * Light <head>-only metadata extraction for URL captures.
 * Fetches first 8KB of HTML, parses title and favicon via regex.
 * Never throws -- returns fallback values on any failure.
 *
 * SSRF protection: only HTTPS URLs allowed, private IP ranges blocked.
 */
```

The function signature:
```typescript
export async function extractHeadMetadata(url: string): Promise<{
  title: string | null;
  favicon_url: string | null;
  estimated_read_time: number | null;
  source_domain: string;
}>
```

Implementation details:
- Parse `source_domain` via `new URL(url).hostname`
- SSRF guard: reject non-HTTPS protocols, block `127.`, `10.`, `192.168.`, `172.16-31.`, `169.254.`, `localhost` via regex on hostname. If blocked, return fallback (null title, source_domain only).
- `fetch(url, { signal: AbortSignal.timeout(3000), headers: { 'User-Agent': 'NeuroGraph/1.0' } })`
- Slice response text to first 8192 chars (head is always near top)
- Title priority: `og:title` meta > `twitter:title` meta > `<title>` tag. Use regex, not a DOM parser.
- Favicon: look for `<link rel="icon">` or `<link rel="shortcut icon">` href. Resolve relative URLs against `https://{source_domain}`. Fallback: `https://{source_domain}/favicon.ico`.
- `estimated_read_time`: always null (deferred to Phase 8 full extraction)
- Wrap entire fetch+parse in try/catch. On ANY failure, return `{ title: null, favicon_url: null, estimated_read_time: null, source_domain }`.

Export a `isSafeUrl` helper function as well (used by the route for early rejection):
```typescript
export function isSafeUrl(raw: string): boolean {
  try {
    const u = new URL(raw);
    if (u.protocol !== 'https:') return false;
    const host = u.hostname;
    if (/^(127\.|10\.|192\.168\.|172\.(1[6-9]|2\d|3[01])\.|169\.254\.)/.test(host)) return false;
    if (host === 'localhost') return false;
    return true;
  } catch {
    return false;
  }
}
```

**Service role client (module-level, outside handler):**

```typescript
import { createClient } from '@supabase/supabase-js';
import type { Database } from '@/types/database';

const supabaseAdmin = createClient<Database>(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!,
  { auth: { autoRefreshToken: false, persistSession: false } }
);
```

**POST handler flow (in exact order):**

1. **Extract bearer token:**
   ```typescript
   const authHeader = request.headers.get('Authorization');
   const rawKeyResult = RawApiKeySchema.safeParse(
     authHeader?.replace(/^Bearer\s+/i, '').trim()
   );
   if (!rawKeyResult.success) {
     return NextResponse.json({ success: false, error: 'unauthorized' }, { status: 401 });
   }
   ```

2. **Hash and look up key:**
   ```typescript
   const keyHash = hashApiKey(rawKeyResult.data);
   const keyRow = await apiKeyQueries.findByHash(supabaseAdmin, keyHash);
   if (!keyRow) {
     return NextResponse.json({ success: false, error: 'unauthorized' }, { status: 401 });
   }
   ```
   Note: `findByHash` already filters `revoked_at IS NULL`. No separate check needed.

3. **Update last_used_at** (fire-and-forget, don't block response):
   ```typescript
   apiKeyQueries.updateLastUsed(supabaseAdmin, keyRow.id).catch(() => {});
   ```

4. **Rate limit check (60 captures/hour):**
   ```typescript
   const oneHourAgo = new Date(Date.now() - 60 * 60 * 1000).toISOString();
   const { count } = await supabaseAdmin
     .from('knowledge_queue')
     .select('*', { count: 'exact', head: true })
     .eq('user_id', keyRow.user_id)
     .gte('created_at', oneHourAgo);

   if ((count ?? 0) >= 60) {
     return NextResponse.json({ success: false, error: 'rate_limited' }, { status: 429 });
   }
   ```

5. **Parse and validate body:**
   ```typescript
   const body = await request.json();
   const parsed = QueueItemInsertSchema.safeParse(body);
   if (!parsed.success) {
     return NextResponse.json({ success: false, error: 'invalid_payload' }, { status: 400 });
   }
   ```

6. **Duplicate URL check (only if URL provided):**
   ```typescript
   if (parsed.data.url) {
     const existing = await queueQueries.findByUrl(supabaseAdmin, keyRow.user_id, parsed.data.url);
     if (existing) {
       return NextResponse.json({
         success: false,
         error: 'duplicate',
         existing_id: existing.id,
       }, { status: 409 });
     }
   }
   ```

7. **URL metadata extraction (only if URL provided and safe):**
   ```typescript
   let meta: { title: string | null; favicon_url: string | null; estimated_read_time: number | null; source_domain: string } | null = null;
   if (parsed.data.url && isSafeUrl(parsed.data.url)) {
     meta = await extractHeadMetadata(parsed.data.url);
   } else if (parsed.data.url) {
     // Unsafe URL (non-HTTPS, private IP) -- still store but skip fetch
     try { meta = { title: null, favicon_url: null, estimated_read_time: null, source_domain: new URL(parsed.data.url).hostname }; } catch { /* invalid URL, meta stays null */ }
   }
   ```

8. **Title resolution (fallback chain):**
   ```typescript
   const title = parsed.data.title ?? meta?.title ?? meta?.source_domain ?? 'Untitled';
   ```
   Note: `QueueItemInsertSchema` requires `title: z.string().min(1)`, but the client may send `title: ""` or rely on URL extraction. The fallback chain handles this. If `parsed.data.title` is an empty string, it fails Zod validation at step 5 and returns `invalid_payload`. No edge case.

9. **Insert queue item:**
   ```typescript
   const item = await queueQueries.create(supabaseAdmin, {
     user_id: keyRow.user_id,
     title,
     url: parsed.data.url ?? null,
     notes: parsed.data.notes ?? null,
     state: 'inbox',
     source_domain: meta?.source_domain ?? null,
     favicon_url: meta?.favicon_url ?? null,
     estimated_read_time: meta?.estimated_read_time ?? null,
   });
   ```

10. **Return success with full item echo:**
    ```typescript
    return NextResponse.json({
      success: true,
      item: {
        id: item.id,
        title: item.title,
        url: item.url,
        source_domain: item.source_domain,
        state: item.state,
      },
    }, { status: 201 });
    ```

**Error handling:** Wrap the entire handler in try/catch. On unexpected errors:
```typescript
catch (error) {
  console.error('[capture] Unexpected error:', error instanceof Error ? error.message : error);
  return NextResponse.json({ success: false, error: 'internal_error' }, { status: 500 });
}
```

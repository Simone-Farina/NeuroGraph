3. **`buildTelemetry` function** — factory for `experimental_telemetry` config objects:
   ```typescript
   export function buildTelemetry(
     agentName: string,
     opts?: { userId?: string; conversationId?: string; extra?: Record<string, string> }
   ): { isEnabled: true; functionId: string; metadata: Record<string, string> } {
     const metadata: Record<string, string> = {};
     if (opts?.userId) metadata.userId = opts.userId;
     if (opts?.conversationId) metadata.conversationId = opts.conversationId;
     if (opts?.extra) Object.assign(metadata, opts.extra);
     return {
       isEnabled: true,
       functionId: agentName,
       metadata,
     };
   }
   ```

   The `functionId` field is what Langfuse uses to name the span. The `agentName` parameter will receive values like `'conversationalist'`, `'architect'`, `'bouncer-extract'`, `'ai-action'`, `'curriculum'`, `'synthesize'`, `'inquisitor'`, `'crystallize-seed'`.

4. **`wrapRagWithObserve` function** — wraps RAG retrieval with Langfuse `observe()` for context logging:
   ```typescript
   import { observe } from '@langfuse/tracing';
   import type { SupabaseClient } from '@supabase/supabase-js';
   import type { Database } from '@/types/database';
   import { getRelevantContext } from '@/lib/ai/rag';

   export async function wrapRagWithObserve(
     query: string,
     userId: string,
     supabase: SupabaseClient<Database>,
     sessionId?: string
   ) {
     return observe(
       {
         name: 'rag-retrieval',
         metadata: {
           userId,
           ...(sessionId ? { sessionId } : {}),
           query,
         },
       },
       () => getRelevantContext(query, userId, supabase)
     );
   }
   ```

    ```typescript
    function renderHighlightedText(
      text: string,
      phrases: string[],
      isHighlighted: boolean,
      highlightClass: string
    ): React.ReactNode {
      if (!isHighlighted || phrases.length === 0) return text;

      // Escape regex metacharacters in each phrase, then build a single capturing-group pattern
      const escaped = phrases.map((p) => p.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));
      const pattern = new RegExp(`(${escaped.join('|')})`, 'gi');
      const parts = text.split(pattern);

      // String.split() with a capturing group places matches at odd indices
      return parts.map((part, i) => {
        const isMatch = i % 2 === 1;
        return isMatch ? (
          <span
            key={i}
            className={`rounded-sm transition-colors duration-300 motion-reduce:transition-none ${highlightClass}`}
          >
            {part}
          </span>
        ) : (
          <span key={i}>{part}</span>
        );
      });
    }
    ```

    Key implementation details:
    - Uses capturing-group regex split — odd-indexed elements are matches (avoids lastIndex drift pitfall from RESEARCH.md)
    - `motion-reduce:transition-none` respects `prefers-reduced-motion` per UI-SPEC accessibility requirement and animate skill
    - `rounded-sm` applies 2px border-radius per UI-SPEC spacing table
    - `transition-colors duration-300` per D-10

    **4. Add `useBloomHighlight` hook:**

    ```typescript
    function useBloomHighlight() {
      const bloomKeyPhrases = useGraphStore((s) => s.bloomKeyPhrases);
      const latestBloomMessageId = useGraphStore((s) => s.latestBloomMessageId);
      const bloomLevel = useGraphStore((s) => s.bloomLevel);
      const [isHighlighted, setIsHighlighted] = useState(false);

      useEffect(() => {
        if (bloomKeyPhrases.length === 0) {
          setIsHighlighted(false);
          return;
        }
        // D-10: fade in when evaluator returns
        setIsHighlighted(true);
        // D-11: fade out after 5 seconds
        const timer = setTimeout(() => setIsHighlighted(false), 5000);
        return () => clearTimeout(timer);
      }, [bloomKeyPhrases]);

      return { isHighlighted, bloomKeyPhrases, latestBloomMessageId, bloomLevel };
    }
    ```

    `isHighlighted` is local component state (not Zustand) because it represents a transient animation state. The `useEffect` cleanup prevents memory leaks on unmount per RESEARCH.md anti-pattern guidance.
